def safe_divide(numerator, denominator):
    try:
        result = numerator / denominator
        return f"SUCCESS: {numerator} / {denominator} = {result:.4f}"
    except ZeroDivisionError as err:
        return f"EXCEPTION CAUGHT: Cannot divide by zero! ({err})"

import string

def generate_secure_password(length=16):
    alphabet = string.ascii_letters + string.digits + "!@#$%^&*()-_=+"
    # Ensure at least one char from each category
    password = [
        secrets.choice(string.ascii_uppercase),
        secrets.choice(string.ascii_lowercase),
        secrets.choice(string.digits),
        secrets.choice("!@#$%^&*()")

    # Fill remainder
    for _ in range(length - 4):

        j = secrets.randbelow(i + 1)
        shuffled[i], shuffled[j] = shuffled[j], shuffled[i]

    return "".join(shuffled)

Estimates the mathematical constant $\pi$ by generating uniform random coordinates inside a unit square.

import random

def estimate_pi(num_darts=100000):
    inside_circle = 0
    for _ in range(num_darts):
        x = random.random()  # [0.0, 1.0)
        y = random.random()  # [0.0, 1.0)
        if (x**2 + y**2) <= 1.0:
            inside_circle += 1
    # Area of quarter circle = pi / 4 => pi = 4 * (inside / total)
    return (4.0 * inside_circle) / num_darts

class ATMConsoleSession:
    def __init__(self, account_holder, initial_balance=10000.0, pin="1234"):
        self.holder = account_holder
        self.balance = initial_balance
        self.pin = pin
        self.transactions = [f"Account Opened: Initial balance INR {initial_balance:,.2f}"]

    def execute_option(self, option_code, entered_pin, amount=0.0):
        if entered_pin != self.pin:
            return "[SECURITY REJECTION] Incorrect PIN entered! Transaction aborted."

        if option_code == 1:
            # Check Balance
            return f"--- BALANCE INQUIRY ---\nAccount: {self.holder}\nAvailable Balance: INR {self.balance:,.2f}"

        elif option_code == 2:
            # Deposit Funds
            if amount <= 0.0:
                return "[DEPOSIT ERROR] Deposit amount must be strictly greater than 0."
            self.balance += amount
            self.transactions.append(f"DEPOSIT: +INR {amount:,.2f} -> Balance: INR {self.balance:,.2f}")
            return f"--- DEPOSIT SUCCESSFUL ---\nDeposited: INR {amount:,.2f}\nUpdated Balance: INR {self.balance:,.2f}"

        elif option_code == 3:
            # Withdraw Funds
            if amount <= 0.0:
                return "[WITHDRAW ERROR] Withdrawal amount must be strictly greater than 0."
            if amount > self.balance:
                return f"[INSUFFICIENT FUNDS] Cannot withdraw INR {amount:,.2f}. Balance: INR {self.balance:,.2f}"
            self.balance -= amount
            self.transactions.append(f"WITHDRAW: -INR {amount:,.2f} -> Balance: INR {self.balance:,.2f}")
            return f"--- WITHDRAWAL SUCCESSFUL ---\nDispensed: INR {amount:,.2f}\nRemaining Balance: INR {self.balance:,.2f}"

        elif option_code == 4:
            # Mini-Statement
            history_lines = "\n  * ".join(self.transactions)
            return f"--- MINI STATEMENT ({self.holder}) ---\n  * {history_lines}\nFinal Net Balance: INR {self.balance:,.2f}"

        elif option_code == 5:
            return f"Thank you for banking with SSM Digital Bank, {self.holder}. Please take your card."

        else:
            return f"[INVALID OPTION] Option {option_code} not recognized. Choose 1, 2, 3, 4, or 5."
